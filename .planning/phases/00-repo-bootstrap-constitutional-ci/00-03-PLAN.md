```typescript
#!/usr/bin/env tsx
import { readFileSync } from 'node:fs';
import { glob } from 'node:fs/promises';
import { exit } from 'node:process';

const CYRILLIC = /[Ѐ-ӿԀ-ԯ]/;
const PATTERNS = [
  '**/*.ts', '**/*.tsx', '**/*.js', '**/*.jsx',
  '**/*.json', '**/*.md', '**/*.mdx',
  '**/*.yaml', '**/*.yml', '**/*.cjs', '**/*.mjs',
];
const IGNORE = [
  '**/node_modules/**', '**/dist/**', '**/coverage/**', '**/.stryker-tmp/**',
  '**/reports/**', '**/.git/**', '**/pnpm-lock.yaml',
  'packages/i18n/locales/**', 'tests/fixtures/i18n/**',
];

const cwd = process.argv[2] ?? process.cwd();
type Offender = { file: string; line: number; col: number; preview: string };
const offenders: Offender[] = [];

async function main() {
  const seen = new Set<string>();
  for (const pattern of PATTERNS) {
    for await (const file of glob(pattern, { cwd, exclude: IGNORE })) {
      if (seen.has(file)) continue;
      seen.add(file);
      const full = `${cwd}/${file}`;
      const lines = readFileSync(full, 'utf8').split('\n');
      lines.forEach((text, i) => {
        const m = CYRILLIC.exec(text);
        if (m) offenders.push({ file, line: i + 1, col: m.index + 1, preview: text.trim().slice(0, 80) });
      });
    }
  }
  if (offenders.length > 0) {
    console.error(`English-only violation: ${offenders.length} occurrence(s)`);
    for (const o of offenders) console.error(`  ${o.file}:${o.line}:${o.col}  ${o.preview}`);
    exit(1);
  }
  console.log(`English-only check passed: ${seen.size} file(s) scanned in ${cwd}`);
}
main().catch((err) => { console.error(err); exit(2); });
```

**tools/lint-tdd.ts** — advisory heuristic (annotation only, non-blocking):
```typescript
#!/usr/bin/env tsx
// Advisory TDD commit-order heuristic.
// Scans the current PR's commits (via gh api or git log when env vars are present).
// Flags any commit modifying production code (apps/api/src/**/*.ts excluding *.test.ts)
// that lands BEFORE any commit modifying a *.test.ts file in the same series.
// Non-zero exit = warning only — CI uses `continue-on-error: true`.
import { execFileSync } from 'node:child_process';
import { exit } from 'node:process';

const baseRef = process.env.GITHUB_BASE_REF ?? 'main';
const headRef = process.env.GITHUB_SHA ?? 'HEAD';

function shellOk(cmd: string, args: string[]): string {
  return execFileSync(cmd, args, { encoding: 'utf8' });
}

try {
  const log = shellOk('git', ['log', '--reverse', '--name-only', '--format=%H', `origin/${baseRef}..${headRef}`]);
  const blocks = log.split(/\n(?=[a-f0-9]{40}\n)/).filter(Boolean);
  let sawTestCommit = false;
  const violations: { sha: string; files: string[] }[] = [];
  for (const block of blocks) {
    const [sha, ...rest] = block.trim().split('\n');
    const files = rest.filter(Boolean);
    const isTest = files.some((f) => /\.test\.ts$/.test(f));
    const prodFiles = files.filter((f) => /^(apps|packages)\/.+\/src\/.+\.ts$/.test(f) && !/\.test\.ts$/.test(f));
    if (isTest) sawTestCommit = true;
    if (!sawTestCommit && prodFiles.length > 0) {
      violations.push({ sha, files: prodFiles });
    }
  }
  if (violations.length > 0) {
    console.warn('TDD heuristic: production-code commit(s) preceded any test commit:');
    for (const v of violations) console.warn(`  ${v.sha} -> ${v.files.join(', ')}`);
    console.warn('Advisory only (non-blocking); promote to blocking in Phase 2+.');
    exit(1);
  }
  console.log('TDD heuristic passed.');
} catch (err) {
  console.error('lint-tdd: error inspecting git log:', err);
  exit(2);
}
```
