**`tests/self-tests/cyrillic-injection.test.ts`:**
```typescript
import { describe, expect, it } from 'vitest';
import { execFileSync } from 'node:child_process';
import { mkdtempSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

describe('DOCS-09 self-test: Cyrillic injection', () => {
  it('makes lint-english.ts exit non-zero with file:line:col diagnostic', () => {
    const root = mkdtempSync(join(tmpdir(), 'cyr-inject-'));
    // Use Unicode escape so this test source itself stays ASCII-only
    const cyrillicGreeting = 'привет'; // privet
    writeFileSync(join(root, 'leak.ts'), `export const x = '${cyrillicGreeting}';\n`);
    let exitCode = 0;
    let stderr = '';
    try {
      execFileSync('pnpm', ['exec', 'tsx', 'tools/lint-english.ts', root], {
        encoding: 'utf8',
        stdio: ['ignore', 'pipe', 'pipe'],
      });
    } catch (err: unknown) {
      const e = err as { status: number; stderr: Buffer };
      exitCode = e.status ?? 1;
      stderr = e.stderr?.toString() ?? '';
    }
    expect(exitCode).not.toBe(0);
    expect(stderr).toMatch(/leak\.ts:1:/);
    expect(stderr).toMatch(/English-only violation/);
  });
});
```

```typescript
import { describe, expect, it } from 'vitest';
import { execFileSync } from 'node:child_process';
import { mkdtempSync, writeFileSync, mkdirSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

describe('TEST-COV-01 self-test: coverage threshold gate', () => {
  it('makes vitest exit non-zero when coverage drops below thresholds', () => {
    const root = mkdtempSync(join(tmpdir(), 'cov-floor-'));
    mkdirSync(join(root, 'src'), { recursive: true });
    // Source with branches; test exercises only one branch -> low coverage
    writeFileSync(join(root, 'src', 'a.ts'),
      `export function classify(n: number): string {
         if (n > 0) return 'pos';
         if (n < 0) return 'neg';
         return 'zero';
       }`);
    writeFileSync(join(root, 'src', 'a.test.ts'),
      `import { describe, expect, it } from 'vitest';
       import { classify } from './a.js';
       describe('classify', () => {
         it('only tests one branch', () => { expect(classify(1)).toBe('pos'); });
       });`);
    writeFileSync(join(root, 'vitest.config.ts'),
      `import { defineConfig } from 'vitest/config';
       export default defineConfig({
         test: {
           coverage: {
             provider: 'v8',
             include: ['src/**/*.ts'],
             exclude: ['**/*.test.ts'],
             thresholds: { lines: 99, branches: 99, functions: 99, statements: 99 },
           },
         },
       });`);
    writeFileSync(join(root, 'package.json'),
      `{"name":"cov-fixture","type":"module","private":true}`);
    let exitCode = 0;
    try {
      execFileSync('pnpm', ['exec', 'vitest', 'run', '--coverage', '--root', root], {
        encoding: 'utf8',
        stdio: ['ignore', 'pipe', 'pipe'],
      });
    } catch (err: unknown) {
      exitCode = (err as { status: number }).status ?? 1;
    }
    expect(exitCode).not.toBe(0);
  }, 60_000);
});
```

```typescript
import { describe, expect, it } from 'vitest';
import { readFileSync } from 'node:fs';
import * as YAML from 'yaml';

const protection = JSON.parse(readFileSync('scripts/branch-protection.json', 'utf8'));
const ci = YAML.parse(readFileSync('.github/workflows/ci.yml', 'utf8'));
const security = YAML.parse(readFileSync('.github/workflows/security.yml', 'utf8'));

const contextNames = new Set<string>(protection.required_status_checks.contexts);
const ciJobs = new Set<string>(Object.keys(ci.jobs ?? {}));
const securityJobs = new Set<string>(Object.keys(security.jobs ?? {}));
const allJobs = new Set<string>([...ciJobs, ...securityJobs]);

describe('CI-03 self-test: branch-protection contexts match actual workflow jobs', () => {
  it('every required context is a real job', () => {
    for (const ctx of contextNames) {
      expect(allJobs.has(ctx), `context ${ctx} not found in ci.yml or security.yml jobs`).toBe(true);
    }
  });

  it('contains the constitutional minimum: lint, typecheck, test, mutation-quick, lint-english, pr-checklist, gitleaks, trivy-fs, codeql, license-scan', () => {
    for (const required of [
      'lint', 'typecheck', 'test', 'mutation-quick', 'lint-english', 'pr-checklist',
      'gitleaks', 'trivy-fs', 'codeql', 'license-scan',
    ]) {
      expect(contextNames.has(required), `${required} missing from contexts`).toBe(true);
    }
  });
});
```
