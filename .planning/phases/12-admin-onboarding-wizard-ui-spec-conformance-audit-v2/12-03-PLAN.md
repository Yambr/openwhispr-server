From RESEARCH §3 (verbatim handler body — implement this contract, EXTENDED for RESEARCH Q1 workspace persistence):
```ts
const claim = await db.execute(sql`
  UPDATE setup_state SET status='completed', completed_at=now()
   WHERE id=1 AND status='pending'
   RETURNING status, completed_at
`);
if (claim.rowCount === 0) {
  const existing = await db.query.users.findFirst({ where: eq(users.role, 'admin') });
  return reply.code(200).send({ admin: { email: existing?.email }, alreadyCompleted: true });
}
const result = await auth.api.signUpEmail({ body: { email, password, name, locale } });
if (result.error) {
  await db.execute(sql`UPDATE setup_state SET status='pending', completed_at=NULL WHERE id=1`);
  return reply.code(400).send({ error: { code: 'ADMIN_CREATE_FAILED', ... } });
}
await db.update(users).set({ role: 'admin' }).where(eq(users.id, result.data.user.id));
// RESEARCH Q1 — persist workspace name to the default tenant singleton.
// Better Auth's signUpEmail runs its own internal transaction (txn-mode PgBouncer),
// so the tenants UPDATE sits OUTSIDE that txn boundary; it is a separate small UPDATE.
// Failure here does NOT roll back the admin user (already created + role assigned);
// instead we surface a warning. See T-12.03-05.
const warnings: string[] = [];
try {
  await db.update(tenants).set({ name: workspace }).where(eq(tenants.id, DEFAULT_TENANT_ID));
} catch (err) {
  req.log.warn({ err, workspace }, 'tenant_rename_failed_after_admin_create');
  warnings.push('tenant_rename_failed');
}
return reply.code(201).send({ admin: { email }, alreadyCompleted: false, ...(warnings.length ? { warnings } : {}) });
```
