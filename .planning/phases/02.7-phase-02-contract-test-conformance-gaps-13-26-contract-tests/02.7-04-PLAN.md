      const url = `${baseUrl}/api/auth/sign-up/email`;
      const res = await fetch(url, { /* ... preserve existing options: method POST, headers, body ... */ });

      const text = await res.text();
      let parsed: { code?: string; message?: string } = {};
      try { parsed = JSON.parse(text) as typeof parsed; } catch { /* non-JSON body */ }

        parsed.code === "USER_ALREADY_EXISTS" ||
        /already exists/i.test(parsed.message ?? "");
      if (isDuplicate) return { created: false };

      );
    }
