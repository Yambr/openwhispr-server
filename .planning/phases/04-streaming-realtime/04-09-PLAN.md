Test client uses the `ws` library (NOT @fastify/websocket — that's server-side):
```
import WebSocket from 'ws';
import { readFileSync } from 'node:fs';

const start = Date.now();
const closeLog = [];
const pingRtts = [];

const ws = new WebSocket('wss://api.localhost:8443/v1/realtime', {
  headers: { authorization: 'Bearer ' + bearer },
  ca: readFileSync('compose/traefik/certs/root-ca.crt'),
});

const pingInterval = setInterval(() => {
  const tStart = Date.now();
  ws.ping('keepalive');
  ws.once('pong', () => pingRtts.push(Date.now() - tStart));
}, 20000);

const responseInterval = setInterval(() => {
  ws.send(JSON.stringify({type: 'response.create'}));
}, 30000);

ws.on('close', (code, reason) => {
  const elapsed = (Date.now() - start) / 1000;
  const isOurs = elapsed < 300 && [1001, 1011].includes(code);
  closeLog.push({t: elapsed, code, reason: reason.toString(), isOurs});
});

await new Promise(resolve => setTimeout(resolve, 305000)); // 5min + 5s margin
clearInterval(pingInterval);
clearInterval(responseInterval);
ws.close(1000);

// Assertions
expect(closeLog.filter(c => c.isOurs)).toEqual([]);
expect(percentile(pingRtts, 0.95)).toBeLessThan(1000);
```

testTimeout for this test: 600000 (10 min — already configured in vitest.e2e.config.ts via plan task 1).
