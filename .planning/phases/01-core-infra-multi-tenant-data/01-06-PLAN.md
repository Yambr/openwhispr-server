    #!/usr/bin/env bash
    # make-backup.sh — pg_dump | age encrypt -> backups/<timestamp>.dump.age
    # Usage: make-backup.sh [<output-dir>]   (default: ./backups)
    # Requires: age in PATH; either docker compose with running postgres OR DATABASE_URL_OWNER env (test mode).
    set -euo pipefail

    OUT_DIR="${1:-${PWD}/backups}"
    PUBKEY_FILE="${PWD}/keys/backup.age.pub"

    if [[ ! -f "${PUBKEY_FILE}" ]]; then
      echo "make-backup: missing ${PUBKEY_FILE} — run tools/bootstrap.sh first" >&2; exit 2
    fi
    if ! command -v age >/dev/null 2>&1; then
      echo "make-backup: age not found in PATH — install via apt/brew/scoop" >&2; exit 2
    fi

    mkdir -p "${OUT_DIR}"
    TS="$(date -u +%Y-%m-%dT%H-%M-%S)"
    OUT="${OUT_DIR}/${TS}.dump.age"
    PUB="$(cat "${PUBKEY_FILE}")"

    if [[ -n "${DATABASE_URL_OWNER:-}" ]]; then
      # test/CI mode: connect directly via PGURL
      pg_dump -Fc "${DATABASE_URL_OWNER}" | age -r "${PUB}" > "${OUT}"
    else
      docker compose exec -T postgres pg_dump -Fc -U "${POSTGRES_OWNER_USER:-openwhispr_owner}" "${POSTGRES_DB:-openwhispr}" \
        | age -r "${PUB}" > "${OUT}"
    fi

    echo "make-backup: wrote ${OUT}"
    ```

    # make-restore.sh — age decrypt | pg_restore. Refuses if target has non-system tables.
    # Usage: BACKUP=path/to/file.dump.age make-restore.sh
    set -euo pipefail

    BACKUP="${BACKUP:-}"
    IDENTITY_FILE="${BACKUP_AGE_IDENTITY_FILE:-${HOME}/.age/key.txt}"

    [[ -n "${BACKUP}" ]] || { echo "make-restore: set BACKUP=path/to/file.dump.age" >&2; exit 1; }
    [[ -f "${BACKUP}" ]] || { echo "make-restore: BACKUP not found: ${BACKUP}" >&2; exit 1; }
    [[ -f "${IDENTITY_FILE}" ]] || { echo "make-restore: identity file not found: ${IDENTITY_FILE} (set BACKUP_AGE_IDENTITY_FILE)" >&2; exit 2; }
    command -v age >/dev/null 2>&1 || { echo "make-restore: age not in PATH" >&2; exit 2; }

    if [[ -n "${DATABASE_URL_OWNER:-}" ]]; then
      # test/CI mode
      COUNT="$(psql "${DATABASE_URL_OWNER}" -tAc "SELECT count(*) FROM information_schema.tables WHERE table_schema='public'")"
      if [[ "${COUNT}" -ne 0 ]]; then
        echo "make-restore: refusing — target has ${COUNT} tables in public schema" >&2; exit 1
      fi
      age -d -i "${IDENTITY_FILE}" "${BACKUP}" | pg_restore -d "${DATABASE_URL_OWNER}"
    else
      COUNT="$(docker compose exec -T postgres psql -U "${POSTGRES_OWNER_USER:-openwhispr_owner}" -d "${POSTGRES_DB:-openwhispr}" -tAc \
        "SELECT count(*) FROM information_schema.tables WHERE table_schema='public'" | tr -d '[:space:]')"
      if [[ "${COUNT}" -ne 0 ]]; then
        echo "make-restore: refusing — target has ${COUNT} tables in public schema" >&2; exit 1
      fi
      age -d -i "${IDENTITY_FILE}" "${BACKUP}" \
        | docker compose exec -T postgres pg_restore -U "${POSTGRES_OWNER_USER:-openwhispr_owner}" -d "${POSTGRES_DB:-openwhispr}"
    fi

    echo "make-restore: ok"
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Operator filesystem -> backups/<file>.age | Encrypted at rest with X25519 recipient |
| GHA secret BACKUP_AGE_IDENTITY -> nightly runner | Restored to ~/.age/key.txt with chmod 600 for the duration of the job |
| docker compose exec -> postgres pg_dump | Runs as openwhispr_owner; full schema + data dumped |
| make-restore.sh -> target Postgres | Refuses non-empty target to prevent accidental clobber |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-06-01 | Information Disclosure | Backup file readable on operator's disk | mitigate | age recipient encryption; only holder of BACKUP_AGE_IDENTITY can decrypt |
| T-01-06-02 | Information Disclosure | age private key committed to repo | mitigate | .gitignore /keys/* with allowlist for backup.age.pub.example only; bootstrap.sh writes private to .env (gitignored); GHA secret in CI |
| T-01-06-03 | Tampering | Restore overwrites live data | mitigate | make-restore.sh refuses if target has tables in public schema |
| T-01-06-04 | Tampering | Backup integrity not verified before restore | accept | age authenticated encryption (ChaCha20-Poly1305) — tampering detected at decrypt time; schema-equivalence diff in nightly catches silent corruption |
| T-01-06-05 | Denial of Service | pg_dump blocks Postgres for large DBs | accept | v1 single-host posture; Phase 9 introduces CNPG with online dumps |
| T-01-06-06 | Information Disclosure | age private key in process listing during restore | mitigate | scripts pass via `-i <file>`, never argv with key contents |
| T-01-06-07 | Spoofing | Backup file substituted with attacker payload | mitigate | age verifies recipient match on decrypt; wrong identity → restore fails |
</threat_model>
