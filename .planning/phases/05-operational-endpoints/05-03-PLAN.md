    `apps/api/src/routes/agent/web-search.ts`:
    ```ts
    app.post('/api/agent/web-search', {
      preHandler: dualAuth,
      config: { rateLimit: { max: 30, timeWindow: '1 minute', keyGenerator: (req) => req.user.id } },
    }, async (req, reply) => {
      const body = WebSearchRequestSchema.parse(req.body);
      const provider = resolveWebSearchProvider();  // resolved at module-load; cached
      if (!provider.isConfigured()) {
        return reply.code(503).send({ error: `${provider.name} not configured (set <ENV_VAR> in .env)` });
      }
      try {
        const result = await provider.search(body.query, body.numResults);
        await withTenant(deps.db, req.tenant, async (tx) => {
          await tx.execute(sql`
            INSERT INTO usage_ledger (tenant_id, user_id, request_id, kind, units)
            VALUES (${req.tenant}::uuid, ${req.user.id}::uuid, ${req.id}, ${`web-search.${provider.name}`}, 1)
            ON CONFLICT (request_id) DO NOTHING
          `);
        });
        return result;
      } catch (e) {
        if (e instanceof MissingProviderKeyError) return reply.code(503).send({ error: e.message });
        if (e instanceof UpstreamError) return reply.code(502).send({ error: 'web-search upstream failed' });
        throw e;  // global error envelope handles
      }
    });
    ```
