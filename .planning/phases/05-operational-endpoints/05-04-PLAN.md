    ```ts
    import { sql } from 'drizzle-orm';

    export async function resolveSttConfig(tx, tenantId: string, userId: string) {
      const [tenantRow, userRow] = await Promise.all([
        tx.execute(sql`SELECT stt_config FROM tenant_settings WHERE tenant_id = ${tenantId}::uuid LIMIT 1`),
        tx.execute(sql`SELECT stt_overrides FROM user_settings WHERE user_id = ${userId}::uuid LIMIT 1`),
      ]);
      const tenantCfg = (tenantRow.rows[0]?.stt_config as Record<string, any>) ?? {};
      const userCfg = (userRow.rows[0]?.stt_overrides as Record<string, any>) ?? {};
      return {
        defaultModel: userCfg.defaultModel ?? tenantCfg.defaultModel ?? process.env.STT_DEFAULT_MODEL ?? 'whisper-1',
        defaultLanguage: userCfg.defaultLanguage ?? tenantCfg.defaultLanguage ?? process.env.STT_DEFAULT_LANGUAGE ?? 'auto',
        availableProviders: computeAvailableProviders(),
      };
    }

    export function computeAvailableProviders(): string[] {
      const out: string[] = [];
      if (process.env.OPENAI_API_KEY) out.push('openai');
      if (process.env.GROQ_API_KEY) out.push('groq');
      if (process.env.ASSEMBLYAI_API_KEY) out.push('assemblyai');
      if (process.env.DEEPGRAM_API_KEY) out.push('deepgram');
      return out;
    }

    export async function resolveNoteRecordingConfig(tx, tenantId: string, userId: string) {
      const [tenantRow, userRow] = await Promise.all([
        tx.execute(sql`SELECT note_recording_config FROM tenant_settings WHERE tenant_id = ${tenantId}::uuid LIMIT 1`),
        tx.execute(sql`SELECT note_recording_overrides FROM user_settings WHERE user_id = ${userId}::uuid LIMIT 1`),
      ]);
      const tenantCfg = (tenantRow.rows[0]?.note_recording_config as Record<string, any>) ?? {};
      const userCfg = (userRow.rows[0]?.note_recording_overrides as Record<string, any>) ?? {};
      const envFormats = (process.env.NOTE_RECORDING_ALLOWED_FORMATS ?? 'webm,ogg,wav,m4a').split(',').map((s) => s.trim()).filter(Boolean);
      return {
        maxDurationSeconds: userCfg.maxDurationSeconds ?? tenantCfg.maxDurationSeconds ?? Number(process.env.NOTE_RECORDING_MAX_DURATION_SECONDS ?? 7200),
        sampleRateHz: userCfg.sampleRateHz ?? tenantCfg.sampleRateHz ?? Number(process.env.NOTE_RECORDING_SAMPLE_RATE_HZ ?? 16000),
        allowedFormats: userCfg.allowedFormats ?? tenantCfg.allowedFormats ?? envFormats,
        diarizationEnabled: userCfg.diarizationEnabled ?? tenantCfg.diarizationEnabled ?? (process.env.NOTE_RECORDING_DIARIZATION_ENABLED !== 'false'),
      };
    }
    ```
