    export interface ParsedListQuery {
      limit: number;
      before: Date | undefined;
      since: Date | undefined;
    }
    export function parseListQuery(q: { limit?: string; before?: string; since?: string }): ParsedListQuery {
      const limit = Math.min(Math.max(parseInt(q.limit ?? '50', 10) || 50, 1), 200);
      const before = q.before ? new Date(q.before) : undefined;
      const since = q.since ? new Date(q.since) : undefined;
      if (before && isNaN(before.getTime())) throw new TypeError('Invalid before timestamp');
      if (since && isNaN(since.getTime())) throw new TypeError('Invalid since timestamp');
      return { limit, before, since };
    }
    ```

    app.post('/api/notes/search', { preHandler: dualAuth }, async (req, reply) => {
      const body = SearchRequestSchema.parse(req.body);  // {query: string min 1 max 256, limit?: number}
      if (body.query.trim().length < 1) return reply.code(400).send({ error: 'query must be non-empty' });
      const limit = Math.min(body.limit ?? 50, 200);
      return withTenant(deps.db, req.tenant, async (tx) => {
        const results = await tx.execute(sql`
          SELECT id, tenant_id, user_id, folder_id, client_note_id, title, content, created_at, updated_at, deleted_at,
                 ts_rank(content_search, query) AS score
          FROM notes, websearch_to_tsquery('simple', ${body.query}) query
          WHERE deleted_at IS NULL
            AND content_search @@ query
          ORDER BY score DESC, created_at DESC
          LIMIT ${limit}
        `);
        return { notes: results.rows };  // per SearchResult shape
      });
    });
    ```
