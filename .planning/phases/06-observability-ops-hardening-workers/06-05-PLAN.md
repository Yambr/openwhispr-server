    ```ts
    import { z } from 'zod';
    import type { PoolClient } from 'pg';
    import { AUDIT_ACTIONS, type AuditAction } from '@openwhispr/data/schema/audit_log';

    const FORBIDDEN_KEYS = ['password','token','bearer','access_token','refresh_token','code','state','virtual_key','api_key','authorization'] as const;

    const baseCtxSchema = z.object({
      tenant_id: z.string().uuid(),
      actor_user_id: z.string().uuid().nullable().optional(),
      request_id: z.string().uuid(),
      ip: z.union([z.string().ip(), z.null()]),
      user_agent: z.string().transform(s => s.slice(0, 512)),
    });

    const methodEnum = z.enum(['password','oauth_google','oauth_github','oauth_microsoft','oauth_okta','email_otp']);
    const signinFailedReason = z.enum(['bad_credentials','expired_otp','oauth_state_mismatch','locked','unknown']);

    export const auditPayloadSchemas = {
      'auth.signin': z.object({ method: methodEnum }),
      'auth.signin_failed': z.object({ method: methodEnum, reason: signinFailedReason }),
      'auth.signout': z.object({}),
      'auth.password_change': z.object({ method: z.enum(['self','admin_forced']) }),
      'auth.oauth_link': z.object({ provider: z.string() }),
      'account.delete': z.object({}),
      'account.delete_requested': z.object({ grace_window_seconds: z.number().int().positive() }),
      'key.issued': z.object({ key_id: z.string() }),
      'key.revoked': z.object({ key_id: z.string(), reason: z.enum(['rotated','manual','compromised']) }),
      'settings.tenant_changed': z.object({ field: z.string(), before_hash: z.string().length(64), after_hash: z.string().length(64) }),
      'settings.user_changed': z.object({ field: z.string(), before_hash: z.string().length(64), after_hash: z.string().length(64) }),
      'admin.tenant_created': z.object({ tenant_id: z.string().uuid() }),
      'admin.tenant_suspended': z.object({ tenant_id: z.string().uuid(), reason: z.string() }),
      'admin.user_impersonated': z.object({ target_user_id: z.string().uuid(), reason: z.string() }),
      'admin.role_changed': z.object({ target_user_id: z.string().uuid(), before: z.string(), after: z.string() }),
      'security.cross_tenant_attempt': z.object({ attempted_tenant_id: z.string().uuid(), route: z.string() }),
      'security.rate_limit_exceeded': z.object({ rule: z.string(), route: z.string() }),
      'security.ssrf_blocked': z.object({ target_url_host: z.string(), rule: z.string(), mode: z.enum(['enforce','warn']).optional() }),
    } as const satisfies Record&lt;AuditAction, z.ZodTypeAny&gt;;

    function rejectForbidden(payload: Record&lt;string, unknown&gt;): void {
      for (const key of Object.keys(payload)) {
        if (FORBIDDEN_KEYS.includes(key.toLowerCase() as never)) {
          throw new Error(`audit payload contains forbidden key: ${key}`);
        }
      }
    }

    export interface AuditCtx { tenant_id: string; actor_user_id?: string | null; request_id: string; ip: string | null; user_agent: string; }

    export async function recordAudit&lt;A extends AuditAction&gt;(
      client: PoolClient,
      ctx: AuditCtx,
      action: A,
      payload: z.infer&lt;typeof auditPayloadSchemas[A]&gt;,
    ): Promise&lt;void&gt; {
      baseCtxSchema.parse(ctx);
      const validated = (auditPayloadSchemas[action] as z.ZodTypeAny).parse(payload);
      rejectForbidden(validated as Record&lt;string, unknown&gt;);
      const ip = process.env.AUDIT_REDACT_IP === 'true' ? null : ctx.ip;
      const ua = ctx.user_agent.slice(0, 512);
      const fullPayload = { ...validated, request_id: ctx.request_id, ip, user_agent: ua };
      await client.query(
        'INSERT INTO audit_log (tenant_id, actor_user_id, action, payload) VALUES ($1, $2, $3, $4)',
        [ctx.tenant_id, ctx.actor_user_id ?? null, action, fullPayload],
      );
    }
    ```
