    ```ts
    import { LRUCache } from 'lru-cache';
    import { request } from 'undici';
    import type { Pool } from 'pg';
    import type { Redis } from 'ioredis';

    export interface DepResult { ok: boolean; latency_ms: number; error?: string; }
    const cache = new LRUCache&lt;string, DepResult&gt;({ max: 16, ttl: 5_000 });
    const inflight = new Map&lt;string, Promise&lt;DepResult&gt;&gt;();

    async function memoized(name: string, fn: () => Promise&lt;DepResult&gt;): Promise&lt;DepResult&gt; {
      const cached = cache.get(name);
      if (cached) return cached;
      const existing = inflight.get(name);
      if (existing) return existing;
      const p = fn().then((r) => { cache.set(name, r); inflight.delete(name); return r; }).catch((e) => { inflight.delete(name); throw e; });
      inflight.set(name, p);
      return p;
    }

    export function makeDepCheck(deps: { pg: Pool; valkey: Redis; litellmUrl: string }) {
      return (name: 'postgres' | 'valkey' | 'litellm') => memoized(name, async () => {
        const start = Date.now();
        try {
          if (name === 'postgres') {
            const c = await deps.pg.connect();
            try { await c.query('SELECT 1'); } finally { c.release(); }
          } else if (name === 'valkey') {
            await deps.valkey.ping();
          } else {
            const { statusCode } = await request(`${deps.litellmUrl}/health`, { bodyTimeout: 2_000, headersTimeout: 2_000 });
            if (statusCode &gt;= 500) throw new Error(`litellm ${statusCode}`);
          }
          return { ok: true, latency_ms: Date.now() - start };
        } catch (err) {
          return { ok: false, latency_ms: Date.now() - start, error: (err as Error).message };
        }
      });
    }
    ```

    ```ts
    import type { FastifyInstance } from 'fastify';
    import os from 'node:os';

    let startupComplete = false;
    export const markStartupComplete = () => { startupComplete = true; };

    export async function registerProbes(app: FastifyInstance, depCheck: ReturnType&lt;typeof import('../lib/dep-check.js').makeDepCheck&gt;) {
      app.get('/livez', { config: { rateLimit: false } }, async () => ({ status: 'ok' }));

      app.get('/readyz', { config: { rateLimit: false } }, async (_req, reply) => {
        const [pg, vk, ll] = await Promise.all([depCheck('postgres'), depCheck('valkey'), depCheck('litellm')]);
        const allOk = pg.ok && vk.ok && ll.ok;
        return reply.code(allOk ? 200 : 503).send({ postgres: pg, valkey: vk, litellm: ll });
      });

      app.get('/startupz', { config: { rateLimit: false } }, async (_req, reply) => {
        return reply.code(startupComplete ? 200 : 503).send({ ready: startupComplete });
      });

      app.get('/api/health', { config: { rateLimit: false } }, async (_req, reply) => {
        reply.header('Deprecation', 'true');
        reply.header('Link', '&lt;/livez&gt;; rel="successor-version"');
        return reply.send({ status: 'ok' });
      });
    }
    ```
