    import { readFileSync, readdirSync } from 'node:fs';
    import path from 'node:path';

    const DASHBOARDS = readdirSync('compose/grafana/provisioning/dashboards').filter(f =&gt; f.endsWith('.json'));

    describe('grafana dashboards validate', () =&gt; {
      for (const file of DASHBOARDS) {
        it(`${file} parses + has required fields`, () =&gt; {
          const json = JSON.parse(readFileSync(path.join('compose/grafana/provisioning/dashboards', file), 'utf8'));
          expect(json.title).toBeTruthy();
          expect(json.schemaVersion).toBeGreaterThanOrEqual(36);
          expect(Array.isArray(json.panels)).toBe(true);
          expect(json.uid).toMatch(/^openwhispr-/);
          for (const p of json.panels) {
            expect(p.datasource?.uid ?? p.datasource).toBeTruthy();
            const hasQuery = p.targets?.some((t: { expr?: string; rawSql?: string; expr_?: string }) =&gt; t.expr || t.rawSql);
            // some panels are text-only — allow but log

          }
        });
      }
    });
