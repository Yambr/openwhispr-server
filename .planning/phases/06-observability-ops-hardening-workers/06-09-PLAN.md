    ```ts
    #!/usr/bin/env tsx
    import { readFileSync } from 'node:fs';
    import { globSync } from 'node:fs';
    import ts from 'typescript';
    import path from 'node:path';

    const ALLOWED_WRAPPERS = new Set(['withTenantContext', 'withSystemContext']);
    const errors: { file: string; line: number; reason: string }[] = [];

    const files = globSync('apps/worker/src/jobs/**/*.ts', { ignore: ['**/*.test.ts', '**/*.d.ts'] });

    for (const file of files) {
      const src = ts.createSourceFile(file, readFileSync(file, 'utf8'), ts.ScriptTarget.ES2022, true);
      let foundDefault = false;
      ts.forEachChild(src, (node) => {
        if (ts.isExportAssignment(node) &amp;&amp; !node.isExportEquals) {
          foundDefault = true;
          const expr = node.expression;
          const isCall = ts.isCallExpression(expr);
          const calleeName = isCall &amp;&amp; ts.isIdentifier(expr.expression) ? expr.expression.text : null;
          if (!isCall || !calleeName || !ALLOWED_WRAPPERS.has(calleeName)) {
            const { line } = src.getLineAndCharacterOfPosition(node.getStart());
            errors.push({ file, line: line + 1, reason: `default export is not a call to withTenantContext/withSystemContext (got ${calleeName ?? expr.kind})` });
          }
        }
      });
      if (!foundDefault) errors.push({ file, line: 0, reason: 'no default export found — every job file MUST default-export a wrapped handler' });
    }

    if (errors.length) {
      for (const e of errors) console.error(`${e.file}:${e.line} ${e.reason}`);
      process.exit(1);
    }
    console.error(`lint:tenant-context — ${files.length} files scanned, all wrapped`);
    process.exit(0);
    ```
