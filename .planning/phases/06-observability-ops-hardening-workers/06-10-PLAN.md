    export const REDACT_PATHS = [
      'req.headers.authorization',
      'req.headers.cookie',
      'req.headers["set-cookie"]',
      'res.headers["set-auth-token"]',
      'res.headers["set-cookie"]',
      '*.token', '*.secret', '*.password', '*.apiKey', '*.api_key',
      '*.virtualKey', '*.virtual_key', '*.client_secret',
      '*.access_token', '*.refresh_token',
      'req.body.password', 'req.body.token', 'req.body.virtual_key',
      'req.query.code', 'req.query.state',
    ];

    export function makePino(opts: { base?: Record&lt;string, unknown&gt; } = {}): pino.Logger {
      return pino({
        base: opts.base,
        redact: { paths: REDACT_PATHS, censor: '[REDACTED]' },
        formatters: {
          level: (label) => ({ level: label }),
        },
        // English-only key constraint enforced by convention + test
      });
    }
