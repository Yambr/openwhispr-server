    import { AsyncLocalStorage } from 'node:async_hooks';
    import { trace } from '@opentelemetry/api';
    import { ownerPool } from '../db/owner-pool.js';
    import { tenantAls } from './with-tenant-context.js';
    import type { Job } from 'bullmq';
    import { z } from 'zod';
    import pino from 'pino';

    const log = pino();
    const tracer = trace.getTracer('worker');

    export function withSystemContext&lt;S extends z.ZodTypeAny | null&gt;(
      schema: S,
      handler: (data: S extends z.ZodTypeAny ? z.infer&lt;S&gt; : Record&lt;string, never&gt;) => Promise&lt;void&gt;,
    ) {
      return async (job: Job) => {
        const data = schema ? schema.parse(job.data) : ({} as never);
        const span = tracer.startSpan(`bullmq.job.${job.queueName}`, { attributes: { mode: 'system', job_id: job.id! } });
        const childLog = log.child({ mode: 'system', job_id: job.id });
        try {
          await tenantAls.run({ tenantId: '__system__', mode: 'system', jobId: job.id! }, async () => {
            await handler(data);
          });
        } catch (err) {
          span.recordException(err as Error);
          childLog.error({ err }, 'system job failed');
          throw err;
        } finally {
          span.end();
        }
      };
    }
    ```

    import type { z } from 'zod';

    export function typedQueue&lt;S extends z.ZodTypeAny&gt;(name: string, schema: S, opts?: ConstructorParameters&lt;typeof Queue&gt;[1]) {
      const q = new Queue(name, opts);
      return {
        underlying: q,
        async add(jobName: string, data: z.infer&lt;S&gt;, jobsOpts?: JobsOptions) {
          const parsed = schema.parse(data);
          return q.add(jobName, parsed, jobsOpts);
        },
        async upsertJobScheduler(schedulerId: string, repeatOpts: Parameters&lt;Queue['upsertJobScheduler']&gt;[1], jobData?: { name: string; data: z.infer&lt;S&gt; }) {
          if (jobData) schema.parse(jobData.data);
          return q.upsertJobScheduler(schedulerId, repeatOpts, jobData as never);
        },
        close: () => q.close(),
      };
    }
    ```

    ```ts
    import { Pool, type PoolClient } from 'pg';
    import { getTenantContext } from '../lib/with-tenant-context.js';

    export class TenantContextMissingError extends Error {
      readonly code = 'TENANT_CONTEXT_MISSING';
      constructor() { super('app.tenant_id GUC not set; wrap handler in withTenantContext or withSystemContext'); }
    }

    export function makeAppPool(connectionString: string): { connect: () => Promise&lt;PoolClient&gt;; end: () => Promise&lt;void&gt; } {
      const pool = new Pool({ connectionString });
      return {
        async connect() {
          const client = await pool.connect();
          let guardChecked = false;
          const originalQuery = client.query.bind(client);
          (client.query as unknown) = (async (...args: unknown[]) => {
            if (!guardChecked) {
              guardChecked = true;
              const ctx = getTenantContext();
              if (ctx?.mode !== 'system') {
                const result = await originalQuery("SELECT current_setting('app.tenant_id', true)::text AS tid");
                const tid = (result as { rows: { tid: string }[] }).rows[0]?.tid ?? '';
                if (!tid) throw new TenantContextMissingError();
              }
            }
            return originalQuery(...(args as Parameters&lt;typeof originalQuery&gt;));
          }) as typeof client.query;
          return client;
        },
        end: () => pool.end(),
      };
    }
    ```

    NOTE: monkey-patching `client.query` requires care because pg's `client.query` has overloads (text|config|callback). The wrapper must preserve all overload signatures. Use a Proxy if direct property assignment proves brittle.
