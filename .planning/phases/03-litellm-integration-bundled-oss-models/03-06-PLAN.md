   Headers: Authorization: Bearer ${PYANNOTE_API_KEY}, Content-Type: application/json
   Body: {url: "media://<key from step 1>"}  (NOT the presigned URL)

4. GET /v1/jobs/{jobId}
   Headers: Authorization: Bearer ${PYANNOTE_API_KEY}
   → 200 {jobId, status: "created"|"running"|"succeeded"|"failed"|"cancelled",
          output?: {duration: number, segments: [{start, end, speaker}, ...]}}

Polling policy: 1500ms interval, 300_000ms hard ceiling.
Billing: pyannote charges only on `succeeded` — orphaned jobs after client disconnect are not billed.

      constructor() { super('PYANNOTE_API_KEY is not configured'); this.name = 'MissingPyannoteKeyError'; }
    }
    export class PyannoteUnavailableError extends Error { name = 'PyannoteUnavailableError'; }
    export class PyannoteAuthError extends Error { name = 'PyannoteAuthError'; }
    export class PyannoteBadRequestError extends Error {
      constructor(public readonly status: number, message: string) { super(message); this.name = 'PyannoteBadRequestError'; }
    }
    export class PyannoteUpstreamError extends Error {
      constructor(public readonly status: number, message: string) { super(message); this.name = 'PyannoteUpstreamError'; }
    }

    export interface PyannoteJob {
      jobId: string;
      status: 'created' | 'running' | 'succeeded' | 'failed' | 'cancelled';
      output?: { duration: number; segments: Array<{ start: number; end: number; speaker: string }> };
    }

    export interface PyannoteClient {
      createMediaInput(): Promise<{ url: string; mediaUri: string }>;
      uploadToPresignedUrl(url: string, body: NodeJS.ReadableStream | Buffer, contentType: string): Promise<void>;
      submitDiarize(mediaUri: string): Promise<string>;            // returns jobId
      pollJob(jobId: string, signal?: AbortSignal): Promise<PyannoteJob>;
    }

    export function createPyannoteClient(opts: {
      apiKey?: string;
      baseUrl?: string;
      dispatcher?: Dispatcher;  // for testing via MockAgent
    } = {}): PyannoteClient {
      const apiKey = opts.apiKey ?? process.env.PYANNOTE_API_KEY;
      if (!apiKey) throw new MissingPyannoteKeyError();
      const baseUrl = opts.baseUrl ?? 'https://api.pyannote.ai';
      const auth = `Bearer ${apiKey}`;

      function classify(status: number, body: string): never {
        if (status === 401 || status === 403) throw new PyannoteAuthError(`pyannote auth failed (${status})`);
        if (status >= 500) throw new PyannoteUnavailableError(`pyannote ${status}`);
        if (status >= 400) throw new PyannoteBadRequestError(status, `pyannote ${status}: ${body.slice(0, 200)}`);
        throw new PyannoteUpstreamError(status, `pyannote unexpected ${status}`);
      }

      return {
        async createMediaInput() {
          const res = await request(`${baseUrl}/v1/media/input`, {
            method: 'POST',
            headers: { authorization: auth, 'content-type': 'application/json' },
            body: '{}',
            dispatcher: opts.dispatcher,
          });
          if (res.statusCode >= 300) classify(res.statusCode, await res.body.text());
          const json = await res.body.json() as { url: string };
          // pyannote returns presigned URL; mediaUri is the media:// reference for /v1/diarize
          // Per pyannote docs the same /v1/media/input call returns both the presigned URL and the mediaUri.
          // If the API response shape evolves, adjust here. As of 2026-05-10 the documented shape is {url}; the
          // mediaUri is the same URL with scheme rewritten OR returned as a separate field — verify at execution
          // by reading the actual response (logged once at startup).
          // Conservative implementation: extract the unique key segment from the presigned URL path and prepend "media://".
          const u = new URL(json.url);
          const key = u.pathname.split('/').filter(Boolean).pop() ?? 'unknown';
          return { url: json.url, mediaUri: `media://${key}` };
        },
        async uploadToPresignedUrl(url, body, contentType) {
          const res = await request(url, {
            method: 'PUT',
            headers: { 'content-type': contentType },
            body,
            dispatcher: opts.dispatcher,
          });
          if (res.statusCode >= 300) {
            const text = await res.body.text();
            throw new PyannoteUpstreamError(res.statusCode, `presigned PUT failed: ${text.slice(0, 200)}`);
          }
        },
        async submitDiarize(mediaUri) {
          const res = await request(`${baseUrl}/v1/diarize`, {
            method: 'POST',
            headers: { authorization: auth, 'content-type': 'application/json' },
            body: JSON.stringify({ url: mediaUri }),
            dispatcher: opts.dispatcher,
          });
          if (res.statusCode >= 300) classify(res.statusCode, await res.body.text());
          const json = await res.body.json() as { jobId: string };
          return json.jobId;
        },
        async pollJob(jobId, signal) {
          if (signal?.aborted) throw new DOMException('aborted', 'AbortError');
          const res = await request(`${baseUrl}/v1/jobs/${encodeURIComponent(jobId)}`, {
            method: 'GET',
            headers: { authorization: auth },
            signal,
            dispatcher: opts.dispatcher,
          });
          if (res.statusCode >= 300) classify(res.statusCode, await res.body.text());
          return await res.body.json() as PyannoteJob;
        },
      };
    }

    import type { Redis } from 'ioredis';

    const KEY_PREFIX = 'diar:idem:';
    const TTL_SECONDS = 86_400;  // 24h

    type Entry = { bodyHash: string; jobId: string | null; createdAt: number };

    export interface IdempotencyCache {
      lookupOrReserve(key: string, bodyHash: string): Promise<
        | { state: 'reserved'; jobId: null }
        | { state: 'hit'; jobId: string }
        | { state: 'in-flight' }
        | { state: 'conflict' }
      >;
      bindJobId(key: string, jobId: string): Promise<void>;
    }

    export function createIdempotencyCache(redis: Redis): IdempotencyCache {
      return {
        async lookupOrReserve(key, bodyHash) {
          const k = KEY_PREFIX + key;
          // SET NX with reservation entry; if NX fails, fetch existing.
          const reservation: Entry = { bodyHash, jobId: null, createdAt: Date.now() };
          const setResult = await redis.set(k, JSON.stringify(reservation), 'EX', TTL_SECONDS, 'NX');
          if (setResult === 'OK') return { state: 'reserved', jobId: null };
          const raw = await redis.get(k);
          if (!raw) return { state: 'reserved', jobId: null };  // race: expired between SET NX and GET; treat as fresh
          const existing = JSON.parse(raw) as Entry;
          if (existing.bodyHash !== bodyHash) return { state: 'conflict' };
          if (existing.jobId) return { state: 'hit', jobId: existing.jobId };
          return { state: 'in-flight' };
        },
        async bindJobId(key, jobId) {
          const k = KEY_PREFIX + key;
          const raw = await redis.get(k);
          if (!raw) {
            // entry expired — re-create with this jobId at full TTL (best-effort)
            await redis.set(k, JSON.stringify({ bodyHash: 'unknown', jobId, createdAt: Date.now() }), 'EX', TTL_SECONDS);
            return;
          }
          const existing = JSON.parse(raw) as Entry;
          existing.jobId = jobId;
          // KEEPTTL preserves the original 24h window so retries within that window still hit.
          await redis.set(k, JSON.stringify(existing), 'KEEPTTL');
        },
      };
    }
    ```

          if (err instanceof MissingPyannoteKeyError) {
            return reply.code(503).send({
              error: 'PYANNOTE_API_KEY is not configured. Set it in .env to enable diarization, or set LITELLM_BASE_URL to a corporate LiteLLM with a single-hop diarization endpoint.',
            });
          }
          throw err;
        }

        // Step 2: read the file part (must support stream; use @fastify/multipart attachFieldsToBody:false).
        // Per Plan 03 Task 2 (HIGH-4), multipart is registered once at buildApp.
        const filePart = await req.file({ limits: { fileSize: MAX_FILE_BYTES } });
        if (!filePart) return reply.code(400).send({ error: 'expected `file` multipart field' });

        // Buffer the file (small for diarization use-cases ≤ ~5min audio); compute SHA-256 for idem fallback.
        // For >5min files we already 504, so buffering here is bounded by MAX_FILE_BYTES limit.
        const chunks: Buffer[] = [];
        let totalBytes = 0;
        const hash = createHash('sha256');
        for await (const chunk of filePart.file) {
          totalBytes += chunk.length;
          if (totalBytes > MAX_FILE_BYTES) {
            return reply.code(400).send({ error: `file exceeds ${MAX_FILE_BYTES} byte limit` });
          }
          hash.update(chunk);
          chunks.push(chunk);
        }
        const fileBuffer = Buffer.concat(chunks);
        const bodyHash = hash.digest('hex');
        const fileMime = filePart.mimetype || 'application/octet-stream';

        // Step 3: idempotency lookup. Idempotency-Key header → fallback to SHA-256(file).
        const headerKey = req.headers['idempotency-key'];
        const idemKey = (typeof headerKey === 'string' && headerKey.length > 0) ? headerKey : bodyHash;

        let jobId: string | null = null;
        const lookup = await idem.lookupOrReserve(idemKey, bodyHash);
        if (lookup.state === 'conflict') {
          return reply.code(409).send({ error: 'Idempotency-Key conflict: same key used with different request body' });
        }
        if (lookup.state === 'hit') {
          jobId = lookup.jobId;
        } else if (lookup.state === 'in-flight') {
          // Brief wait + re-check (in-flight = another request just reserved but hasn't bound jobId yet).
          for (let i = 0; i < 4; i++) {
            await sleep(250);
            const recheck = await idem.lookupOrReserve(idemKey, bodyHash);
            if (recheck.state === 'hit') { jobId = recheck.jobId; break; }
            if (recheck.state === 'conflict') return reply.code(409).send({ error: 'Idempotency-Key conflict' });
          }
          if (!jobId) {
            reply.header('Retry-After', '5');
            return reply.code(503).send({ error: 'concurrent request in flight; retry shortly' });
          }
        }

        // Step 4-6: if no cached jobId, orchestrate the upload + submit flow.
        if (!jobId) {
          try {
            const { url: presignedUrl, mediaUri } = await pyannote.createMediaInput();
            await pyannote.uploadToPresignedUrl(presignedUrl, fileBuffer, fileMime);
            jobId = await pyannote.submitDiarize(mediaUri);
            await idem.bindJobId(idemKey, jobId);
          } catch (err) {
            return mapPyannoteError(err, reply);
          }
        }

        // Step 7-9: poll loop + abort-on-disconnect.
        const abortController = new AbortController();
        const onClose = () => abortController.abort();
        req.raw.on('close', onClose);

        const startedAt = Date.now();
        try {
          while (Date.now() - startedAt < POLL_CEILING_MS) {
            if (abortController.signal.aborted) {
              // Client disconnected — pyannote job continues; idem cache holds jobId for retry.
              req.log.info({ jobId }, 'diarization client disconnected during poll');
              return;  // no response — connection already gone
            }
            let job;
            try {
              job = await pyannote.pollJob(jobId, abortController.signal);
            } catch (err) {
              if ((err as Error).name === 'AbortError') return;
              return mapPyannoteError(err, reply);
            }
            if (job.status === 'succeeded' && job.output) {
              return reply.code(200).send(DiarizationResponse.parse(job.output));
            }
            if (job.status === 'failed' || job.status === 'cancelled') {
              return reply.code(502).send({ error: `diarization job ${job.status}`, jobId });
            }
            await sleep(POLL_INTERVAL_MS, undefined, { signal: abortController.signal }).catch(() => {});
          }
          // Step 12: ceiling exceeded.
          return reply.code(504).send({
            error: 'diarization exceeded 5-minute ceiling; for files > 5min consider corporate LiteLLM override with self-hosted Speaches',
            jobId,
          });
        } finally {
          req.raw.off('close', onClose);
        }
      };
    }

    function mapPyannoteError(err: unknown, reply: FastifyReply) {
      if (err instanceof PyannoteUnavailableError) {
        reply.header('Retry-After', '30');
        return reply.code(503).send({ error: 'pyannote.ai upstream unavailable' });
      }
      if (err instanceof PyannoteAuthError) {
        return reply.code(503).send({ error: 'PYANNOTE_API_KEY rejected by upstream — verify the key in .env' });
      }
      if (err instanceof PyannoteBadRequestError) {
        return reply.code(502).send({ error: `pyannote rejected request (${err.status})` });
      }
      throw err;
    }

    export default buildDiarizationRoutes;
    ```
