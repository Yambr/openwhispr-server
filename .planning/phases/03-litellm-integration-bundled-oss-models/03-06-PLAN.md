       ```typescript
       import { type LitellmClient, MissingProviderKeyError, LitellmUpstreamError } from '@openwhispr/litellm-client';
       import type { FastifyInstance } from 'fastify';
       import { DiarizationResponse } from '@openwhispr/contract-tests/schemas';

       export interface DiarizationDeps {
         litellm: LitellmClient;
         /** Set via env MOCK_DIARIZATION=true in contract-test profile (research §Pattern 4 note). */
         mockMode?: boolean;
       }

       const MOCK_RESPONSE = {
         segments: [
           { start: 0.0, end: 1.0, speaker: 'SPEAKER_00' },
         ],
       };

       export const buildDiarizationRoutes = (deps: DiarizationDeps) =>
         async function diarizationRoutes(app: FastifyInstance): Promise<void> {
           // Mount path(s) per Plan 01 / docs/wire-contracts-phase-3.md.
           // Adjust the URL string(s) to match the locked decision.
           const MOUNT_PATHS = ['/v1/audio/diarization']; // Plan 01 may add '/api/diarization'

           for (const url of MOUNT_PATHS) {
             app.route({
               method: 'POST',
               url,
               config: { rateLimit: { max: 30, timeWindow: '1 minute' } },
               handler: async (req, reply) => {
                 if (!req.user || !req.tenant) return reply.code(401).send({ error: 'unauthorized' });

                 if (deps.mockMode) {
                   return reply.code(200).send(DiarizationResponse.parse(MOCK_RESPONSE));
                 }

                 const contentType = req.headers['content-type'];
                 if (!contentType) return reply.code(400).send({ error: 'expected multipart audio' });

                 // Bundled-default provider-key gate (skip when LITELLM_BASE_URL is overridden).
                 const isOverride = !!process.env.LITELLM_BASE_URL && process.env.LITELLM_BASE_URL !== 'http://litellm:4000';
                 if (!isOverride && !process.env.PYANNOTE_API_KEY && !process.env.DIARIZATION_API_KEY) {
                   return reply.code(503).send({
                     error:
                       'PYANNOTE_API_KEY (or DIARIZATION_API_KEY) is not configured. Set it in .env to enable diarization, ' +
                       'or set LITELLM_BASE_URL to a corporate LiteLLM with a diarization pass_through endpoint.',
                   });
                 }

                 try {
                   const upstream = await deps.litellm.passthrough('/v1/audio/diarization', {
                     method: 'POST',
                     body: req.raw,
                     contentType,
                     userId: req.user.id,
                     requestId: req.id,
                   });
                   const json = await upstream.body.json();
                   return reply.code(upstream.statusCode).send(json);
                 } catch (err) {
                   if (err instanceof MissingProviderKeyError) return reply.code(503).send({ error: err.message });
                   if (err instanceof LitellmUpstreamError) {
                     req.log.warn({ status: err.status }, 'litellm upstream error on diarization');
                     return reply.code(502).send({ error: 'upstream diarization provider failure' });
                   }
                   throw err;
                 }
               },
             });
           }
         };

       export default buildDiarizationRoutes;
       ```

       Note: NO usage_ledger write — diarization spend is unmetered through pass-through (LITELLM-07 acknowledges; documented in 03-RESEARCH.md). Spend ingest worker (Plan 08) will pick up rows IF LiteLLM logs them; we do not duplicate-write app-side.
