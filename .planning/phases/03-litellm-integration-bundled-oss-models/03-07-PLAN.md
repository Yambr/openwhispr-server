       ```typescript
       import fastifyHttpProxy from '@fastify/http-proxy';
       import type { FastifyInstance } from 'fastify';
       import type { LitellmClient } from '@openwhispr/litellm-client';
       import { AuthError } from '../errors.js';

       export interface RealtimeDeps {
         litellm: LitellmClient;
         /** Master key passed explicitly so tests can inject without env. */
         masterKey: string;
       }

       export const buildRealtimeRoutes = (deps: RealtimeDeps) =>
         async function realtimeRoutes(app: FastifyInstance): Promise<void> {
           const upstreamHttp = deps.litellm.baseUrl;                       // http://litellm:4000
           const upstreamWs = upstreamHttp.replace(/^http(s)?:/, 'ws$1:');  // ws://litellm:4000

           await app.register(fastifyHttpProxy, {
             upstream: upstreamHttp,
             wsUpstream: upstreamWs,
             prefix: '/v1/realtime',
             rewritePrefix: '/v1/realtime',
             websocket: true,
             wsClientOptions: {
               // Strip desktop's opaque bearer; inject LiteLLM master key.
               // Add request_id metadata header (Pitfall #6 / D-08).
               rewriteRequestHeaders: (headers, request) => {
                 const userId = (request as any).user?.id ?? 'anonymous';
                 const requestId = request.id;
                 return {
                   ...headers,
                   authorization: `Bearer ${deps.masterKey}`,
                   'x-litellm-spend-logs-metadata': JSON.stringify({
                     openwhispr_request_id: requestId,
                     openwhispr_user_id: userId,
                   }),
                 };
               },
             },
             preHandler: async (req, _reply) => {
               // dualAuthHook runs as the global preHandler and would have
               // attached req.user already. Defensive: re-check here so we
               // never upgrade for an unauthenticated request.
               if (!(req as any).user) throw new AuthError('unauthorized');
               // Append ?user=<userId> for LiteLLM end_user attribution (D-03).
               // Fastify exposes raw URL on req.raw.url; mutate to inject query.
               const u = new URL(req.url, 'http://internal');
               u.searchParams.set('user', (req as any).user.id);
               req.raw.url = u.pathname + u.search;
             },
           });
         };

       export default buildRealtimeRoutes;
       ```
