    4. `packages/litellm-client/src/index.ts`:
       ```typescript
       import { request as undiciRequest, type Dispatcher } from 'undici';
       import type { Readable } from 'node:stream';
       import { MissingProviderKeyError, LitellmUpstreamError } from './errors.js';
       import type { LitellmClientConfig } from './config.js';

       // Static map of bundled-default model alias → provider-key env var.
       // Mirrors compose/litellm/litellm_config.yaml model_list. If the
       // operator overrides LITELLM_BASE_URL to a corporate proxy, this map
       // is bypassed (corporate operator owns its own auth posture).
       const BUNDLED_MODEL_PROVIDER: Record<string, keyof LitellmClientConfig['providerKeys']> = {
         'qwen3.5-plus': 'openrouter',
         'gemini-3-flash': 'openrouter',
         'gpt-4o-mini': 'openrouter',
         'whisper-1': 'openai',
       };
       const PROVIDER_ENV_VAR = {
         openrouter: 'OPENROUTER_API_KEY',
         openai: 'OPENAI_API_KEY',
         pyannote: 'PYANNOTE_API_KEY',
       } as const;

       export interface ChatCompletionRequest {
         model?: string;
         messages: Array<{ role: string; content: string }>;
         userId: string;
         requestId: string;
         /** Pass-through additional OpenAI chat-completion params. */
         extras?: Record<string, unknown>;
       }

       export interface LitellmClient {
         chatCompletions(req: ChatCompletionRequest): Promise<Dispatcher.ResponseData>;
         audioTranscriptions(args: {
           body: Readable;
           contentType: string;
           userId: string;
           requestId: string;
         }): Promise<Dispatcher.ResponseData>;
         passthrough(path: string, args: {
           method: string;
           body?: Readable | string | Buffer;
           contentType?: string;
           userId: string;
           requestId: string;
         }): Promise<Dispatcher.ResponseData>;
         /** Test seam: lets routes inspect baseUrl for proxy mounts (Plan 06 wsUpstream). */
         readonly baseUrl: string;
       }

       export function buildLitellmClient(config: LitellmClientConfig, isOverride = !!process.env.LITELLM_BASE_URL): LitellmClient {
         function checkProviderKey(model: string) {
           // When LITELLM_BASE_URL points at a corporate proxy, skip the local
           // provider-key check (corporate proxy owns its own auth).
           if (isOverride) return;
           const provider = BUNDLED_MODEL_PROVIDER[model];
           if (!provider) return; // unknown model: let upstream fail
           if (!config.providerKeys[provider]) {
             throw new MissingProviderKeyError(PROVIDER_ENV_VAR[provider], model);
           }
         }

         function authHeaders(userId: string, requestId: string): Record<string, string> {
           return {
             authorization: `Bearer ${config.masterKey}`,
             'x-litellm-end-user-id': userId,
             'x-litellm-spend-logs-metadata': JSON.stringify({ openwhispr_request_id: requestId }),
           };
         }

         return {
           baseUrl: config.baseUrl,
           async chatCompletions(req) {
             const model = req.model ?? config.defaultChatModel;
             checkProviderKey(model);
             const body = JSON.stringify({
               ...req.extras,
               model,
               messages: req.messages,
               user: req.userId, // D-03: per-user attribution via OpenAI-compatible field
             });
             const res = await undiciRequest(`${config.baseUrl}/v1/chat/completions`, {
               method: 'POST',
               headers: { ...authHeaders(req.userId, req.requestId), 'content-type': 'application/json' },
               body,
             });
             if (res.statusCode >= 400) {
               const bodyText = await res.body.text();
               throw new LitellmUpstreamError(res.statusCode, bodyText);
             }
             return res;
           },
           async audioTranscriptions(args) {
             checkProviderKey('whisper-1');
             const res = await undiciRequest(`${config.baseUrl}/v1/audio/transcriptions`, {
               method: 'POST',
               headers: { ...authHeaders(args.userId, args.requestId), 'content-type': args.contentType },
               body: args.body,
             });
             if (res.statusCode >= 400) {
               const bodyText = await res.body.text();
               throw new LitellmUpstreamError(res.statusCode, bodyText);
             }
             return res;
           },
           async passthrough(path, args) {
             const headers: Record<string, string> = authHeaders(args.userId, args.requestId);
             if (args.contentType) headers['content-type'] = args.contentType;
             const res = await undiciRequest(`${config.baseUrl}${path}`, {
               method: args.method as Dispatcher.HttpMethod,
               headers,
               body: args.body,
             });
             if (res.statusCode >= 400) {
               const bodyText = await res.body.text();
               throw new LitellmUpstreamError(res.statusCode, bodyText);
             }
             return res;
           },
         };
       }

       export type { LitellmClientConfig } from './config.js';
       export { loadLitellmConfigFromEnv } from './config.js';
       export { MissingProviderKeyError, LitellmUpstreamError } from './errors.js';
       ```
