    ```typescript
    import { ReasonRequest, ReasonResponse } from '@openwhispr/contract-tests/schemas';
    import { type ExecutableTx, type TransactionalDb, withTenant } from '@openwhispr/data';
    import {
      type LitellmClient,
      MissingProviderKeyError,
      LitellmUpstreamError,
    } from '@openwhispr/litellm-client';
    import { sql } from 'drizzle-orm';
    import type { FastifyInstance } from 'fastify';

    export interface ReasonDeps {
      db: TransactionalDb<ExecutableTx>;
      litellm: LitellmClient;
      /** Optional override for default model — useful in tests; production picks up from litellm-client config (D-06). */
      defaultModel?: string;
    }

    /**
     * Static map: model alias → provider name returned in ReasonResponse.provider.
     * Mirrors compose/litellm/litellm_config.yaml (Plan 01) + BUNDLED_MODEL_PROVIDER
     * in litellm-client. When LITELLM_BASE_URL points at a corporate proxy, the
     * operator's mapping may differ; we fall back to 'litellm' as the generic provider.
     */
    const MODEL_PROVIDER: Record<string, string> = {
      'qwen3.5-plus': 'openrouter',
      'gemini-3-flash': 'openrouter',
      'gpt-4o-mini': 'openrouter',
    };

    export const buildReasonRoutes = (deps: ReasonDeps) =>
      async function reasonRoutes(app: FastifyInstance): Promise<void> {
        app.route({
          method: 'POST',
          url: '/api/reason',
          config: { rateLimit: { max: 120, timeWindow: '1 minute' } },
          schema: { body: ReasonRequest },
          handler: async (req, reply) => {
            if (!req.user || !req.tenant) return reply.code(401).send({ error: 'unauthorized' });
            const body = ReasonRequest.parse(req.body);
            const model = body.model ?? deps.defaultModel ?? 'qwen3.5-plus';
            try {
              const upstream = await deps.litellm.chatCompletions({
                model,
                messages: [{ role: 'user', content: body.text }],
                userId: req.user.id,
                requestId: req.id,
              });
              const upstreamJson = await upstream.body.json() as {
                model: string;
                choices: Array<{ message: { role: string; content: string } }>;
                usage?: { total_tokens: number };
              };
              const tokens = upstreamJson.usage?.total_tokens ?? 0;

              await withTenant(deps.db, req.tenant, async (tx) => {
                await tx.execute(sql`
                  INSERT INTO usage_ledger (tenant_id, user_id, request_id, kind, units)
                  VALUES (${req.tenant}::uuid, ${req.user!.id}::uuid, ${req.id}, 'reason_tokens', ${tokens})
                  ON CONFLICT (request_id) DO NOTHING
                `);
              });

              const response: ReasonResponse = {
                text: upstreamJson.choices[0]?.message?.content ?? '',
                model: upstreamJson.model ?? model,
                provider: MODEL_PROVIDER[model] ?? 'litellm',
                promptMode: body.promptMode ?? 'default',
                matchType: body.matchType ?? 'default',
              };
              return reply.code(200).send(ReasonResponse.parse(response));
            } catch (err) {
              if (err instanceof MissingProviderKeyError) return reply.code(503).send({ error: err.message });
              if (err instanceof LitellmUpstreamError) {
                req.log.warn({ status: err.status }, 'litellm upstream error on /api/reason');
                return reply.code(502).send({ error: 'upstream reasoning provider failure' });
              }
              throw err;
            }
          },
        });
      };

    export default buildReasonRoutes;
    ```

| T-03-05-02 | Information Disclosure | Prompt text logged at INFO level | mitigate | req.log scrubs body fields; only model + status logged at WARN/ERROR. |
| T-03-05-03 | DoS | Large prompt → expensive token bill | accept | v1 has no quota enforcement (DATA-03 observability-only); rate-limit 120/min/user. v2 quota work owns enforcement. |

| T-03-05-05 | Confused deputy | usage_ledger row writes other tenant's row | mitigate | withTenant SET LOCAL app.tenant_id; RLS enforced. |
