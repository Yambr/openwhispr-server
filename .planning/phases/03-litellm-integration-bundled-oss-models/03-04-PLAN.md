       ```typescript
       import { TranscribeResponse } from '@openwhispr/contract-tests/schemas';
       import { type ExecutableTx, type TransactionalDb, withTenant } from '@openwhispr/data';
       import {
         type LitellmClient,
         MissingProviderKeyError,
         LitellmUpstreamError,
       } from '@openwhispr/litellm-client';
       import { sql } from 'drizzle-orm';
       import type { FastifyInstance } from 'fastify';
       import { minutesFromDuration } from '../lib/word-units.js';

       export interface TranscribeDeps {
         db: TransactionalDb<ExecutableTx>;
         litellm: LitellmClient;
       }

       export const buildTranscribeRoutes = (deps: TranscribeDeps) =>
         async function transcribeRoutes(app: FastifyInstance): Promise<void> {
           // Important: do NOT use the default @fastify/multipart body parsing.
           // We forward `req.raw` directly to LiteLLM via undici streaming
           // (Pitfall #5). @fastify/multipart MUST be registered with
           // attachFieldsToBody:false at the buildApp level (Plan owns
           // verifying this — assert at startup if needed).
           app.route({
             method: 'POST',
             url: '/api/transcribe',
             config: { rateLimit: { max: 60, timeWindow: '1 minute' } },
             // No `schema.body` — multipart streaming bypasses the JSON body
             // parser. Response shape validated by zod below.
             handler: async (req, reply) => {
               if (!req.user || !req.tenant) {
                 // dualAuthHook should have thrown by now; defensive
                 return reply.code(401).send({ error: 'unauthorized' });
               }

               const contentType = req.headers['content-type'];
               if (!contentType || !contentType.toLowerCase().startsWith('multipart/form-data')) {
                 return reply.code(400).send({ error: 'expected multipart/form-data audio upload' });
               }

               try {
                 const upstream = await deps.litellm.audioTranscriptions({
                   body: req.raw,
                   contentType,
                   userId: req.user.id,
                   requestId: req.id,
                 });

                 // Parse upstream JSON. Whisper default response_format=json gives
                 // {text}; verbose_json gives {text,duration,language,segments}.
                 const upstreamJson = await upstream.body.json() as {
                   text: string;
                   duration?: number;
                   language?: string;
                   segments?: unknown[];
                 };

                 const minutes = minutesFromDuration(upstreamJson.duration);

                 // DATA-03: idempotent ledger insert. Worker also writes from spend
                 // logs (Plan 08) — both UPSERTs converge on the same row.
                 await withTenant(deps.db, req.tenant, async (tx) => {
                   await tx.execute(sql`
                     INSERT INTO usage_ledger (tenant_id, user_id, request_id, kind, units)
                     VALUES (${req.tenant}::uuid, ${req.user!.id}::uuid, ${req.id}, 'transcribe_minutes', ${minutes})
                     ON CONFLICT (request_id) DO NOTHING
                   `);
                 });

                 const response: TranscribeResponse = {
                   text: upstreamJson.text,
                   wordsUsed: minutes, // Plan 01 decision: minutes-of-audio
                   wordsRemaining: 999_999_999, // unlimited in v1
                   plan: 'unlimited',
                   limitReached: false,
                   sttProvider: 'openai',
                   sttModel: 'whisper-1',
                   language: upstreamJson.language,
                   duration: upstreamJson.duration,
                   segments: upstreamJson.segments,
                 };
                 return reply.code(200).send(TranscribeResponse.parse(response));
               } catch (err) {
                 if (err instanceof MissingProviderKeyError) {
                   return reply.code(503).send({ error: err.message });
                 }
                 if (err instanceof LitellmUpstreamError) {
                   req.log.warn({ status: err.status }, 'litellm upstream error on /api/transcribe');
                   return reply.code(502).send({ error: 'upstream transcription provider failure' });
                 }
                 throw err; // unknown — let centralized error handler emit canonical envelope
               }
             },
           });
         };

       export default buildTranscribeRoutes;
       ```

       export function audioMultipartBody(filename = 'sample-1s.wav'): { body: Buffer; contentType: string } {

<threat_model>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| desktop ↔ api `/api/transcribe` | Bearer or cookie auth required; multipart body untrusted |
| api ↔ litellm | LITELLM_MASTER_KEY-injected; userId from req.user (not request body) |
| api ↔ postgres usage_ledger | RLS-enforced via withTenant; request_id UNIQUE prevents duplicate writes |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-04-01 | Denial of Service | Multipart body unbounded → memory exhaustion | mitigate | @fastify/multipart limits.fileSize=100MB; rate-limit 60/min/user. Beyond 100MB → 413. |
| T-03-04-02 | Information Disclosure | Upstream error body leaks LITELLM_MASTER_KEY | mitigate | LitellmUpstreamError truncates body to 200 chars in client; route never echoes upstream body to client (returns generic 502 envelope). |
| T-03-04-03 | Spoofing | Caller injects `user` body field to attribute spend to other user | mitigate | Client overrides `user` to req.user.id (set by dualAuthHook); caller-supplied field discarded. |
| T-03-04-04 | Repudiation | usage_ledger writes lost on api crash | mitigate | DATA-03 idempotency + Plan 08 spend ingest worker also writes from spend logs → cross-write convergence. |
| T-03-04-05 | Tampering | Multipart Content-Type boundary mismatch | accept | Forwarded as-is; LiteLLM/upstream Whisper validates and 4xx's; mapped to 502 envelope. |
| T-03-04-06 | Authorization bypass | 503 on missing key looks like auth failure → desktop signs out | mitigate | Pitfall #8: 503 (not 401) for config issues; WIRE-18 (401-means-session-expired) preserved. |
