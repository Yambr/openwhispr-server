       **CRITICAL — GitHub Actions restriction (CRIT-1 fix):** the `secrets` context is NOT
       available in job-level `if:` conditions. Using `if: ${{ secrets.X != '' }}` at job level
       fails workflow validation OR silently skips every run. Gate via a preceding step that

       ```yaml
       e2e-test:
         name: E2E (real provider APIs)
         runs-on: ubuntu-latest
         # NOTE: NO `if: ${{ secrets.* }}` here — forbidden at job level.
         steps:
           - uses: actions/checkout@<sha-pinned>
           - name: Check provider-key secret presence
             id: gate
             env:
               OPENROUTER_KEY: ${{ secrets.OPENROUTER_API_KEY }}
             run: |
               if [ -n "$OPENROUTER_KEY" ]; then
                 echo "have_keys=true" >> "$GITHUB_OUTPUT"
               else
                 echo "have_keys=false" >> "$GITHUB_OUTPUT"
                 echo "::notice::OPENROUTER_API_KEY not configured — skipping E2E."
               fi
           - name: Setup Node
             if: steps.gate.outputs.have_keys == 'true'
             uses: actions/setup-node@<sha-pinned>
             with: { node-version: 24 }
           - name: Install pnpm
             if: steps.gate.outputs.have_keys == 'true'
             uses: pnpm/action-setup@<sha-pinned>
             with: { version: 10 }
           - name: pnpm install
             if: steps.gate.outputs.have_keys == 'true'
             run: pnpm install --frozen-lockfile
           - name: Bootstrap secrets
             if: steps.gate.outputs.have_keys == 'true'
             run: bash tools/bootstrap.sh --ci
           - name: Write .env.e2e
             if: steps.gate.outputs.have_keys == 'true'
             env:
               OPENROUTER_API_KEY: ${{ secrets.OPENROUTER_API_KEY }}
               OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
               PYANNOTE_API_KEY: ${{ secrets.PYANNOTE_API_KEY }}
             run: |
               # Use env-var indirection (not direct ${{ secrets.* }} expansion in the
               # script body) so GHA's secret-masking applies and avoids quoting pitfalls.
               printf 'OPENROUTER_API_KEY=%s\n' "$OPENROUTER_API_KEY" >> .env.e2e
               printf 'OPENAI_API_KEY=%s\n' "$OPENAI_API_KEY" >> .env.e2e
               printf 'PYANNOTE_API_KEY=%s\n' "$PYANNOTE_API_KEY" >> .env.e2e
           - name: Run E2E
             if: steps.gate.outputs.have_keys == 'true'
             run: make e2e-test
       ```
