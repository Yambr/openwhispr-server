    1. `apps/worker/src/jobs/ingest-litellm-spend.ts` (template per research §Code Example 3, with corrections from Plan 02 spike):
       ```typescript
       import { Queue, Worker, type ConnectionOptions } from 'bullmq';
       import type { Pool } from 'pg';
       import pino from 'pino';
       import { inferKind } from '../lib/infer-kind.js';

       const log = pino({ name: 'ingest-litellm-spend' });
       const QUEUE_NAME = 'litellm-spend-ingest';
       const SCHEDULER_KEY = 'ingest-litellm-spend';
       const WATERMARK_KEY = 'litellm:spend:last_start_time';
       const BATCH_SIZE = 1000;

       export interface JobDeps {
         litellmPool: Pool;
         appOwnerPool: Pool;
         connection: ConnectionOptions;
       }

       export function createQueue(connection: ConnectionOptions): Queue {
         return new Queue(QUEUE_NAME, { connection });
       }

       export async function ensureScheduler(queue: Queue): Promise<void> {
         await queue.upsertJobScheduler(
           SCHEDULER_KEY,
           { every: 30_000 },
           { name: 'ingest', data: {} },
         );
       }

       export async function runIngestOnce(deps: JobDeps): Promise<{ rowsProcessed: number }> {
         const redis = (deps.connection as any).client; // BullMQ exposes client when connection is a redis instance
         // Determine watermark — read from a small key/value store. Use a dedicated
         // ioredis client for simplicity; in production code, share with BullMQ connection.
         const since = (await redis.get?.(WATERMARK_KEY)) ?? new Date(Date.now() - 5 * 60_000).toISOString();
         const { rows } = await deps.litellmPool.query<{
           request_id: string;
           end_user: string | null;
           total_tokens: number | null;
           model: string;
           startTime: string;
           metadata: Record<string, any> | null;
         }>(`
           SELECT request_id, "end_user", total_tokens, model, "startTime", metadata
           FROM "LiteLLM_SpendLogs"
           WHERE "startTime" > $1
           ORDER BY "startTime" ASC
           LIMIT ${BATCH_SIZE}
         `, [since]);

         let processed = 0;
         for (const r of rows) {
           const ourRid = r.metadata?.openwhispr_request_id ?? r.request_id;
           const userId = r.end_user;
           if (!userId) {
             log.warn({ rid: ourRid }, 'spend log missing end_user — skipping');
             continue;
           }
           // Resolve tenant via users table (owner pool — BYPASSRLS).
           const tenantRes = await deps.appOwnerPool.query<{ tenant_id: string }>(
             `SELECT tenant_id FROM users WHERE id = $1::uuid LIMIT 1`,
             [userId],
           );
           const tenantId = tenantRes.rows[0]?.tenant_id;
           if (!tenantId) {
             log.warn({ rid: ourRid, userId }, 'no tenant for user — skipping');
             continue;
           }
           const kind = inferKind(r.model);
           const units = kind === 'reason_tokens'
             ? (r.total_tokens ?? 0)
             : Math.ceil((r.metadata?.duration ?? 0) / 60);
           await deps.appOwnerPool.query(`
             INSERT INTO usage_ledger (tenant_id, user_id, request_id, kind, units)
             VALUES ($1::uuid, $2::uuid, $3, $4, $5)
             ON CONFLICT (request_id) DO NOTHING
           `, [tenantId, userId, ourRid, kind, units]);
           processed++;
         }

         if (rows.length > 0) {
           await redis.set?.(WATERMARK_KEY, rows[rows.length - 1].startTime);
         }
         return { rowsProcessed: processed };
       }

       export function createWorker(deps: JobDeps): Worker {
         return new Worker(
           QUEUE_NAME,
           async () => runIngestOnce(deps),
           { connection: deps.connection },
         );
       }
       ```

       import { createQueue, ensureScheduler, createWorker } from './jobs/ingest-litellm-spend.js';
       import { makeLitellmPool } from './db/litellm-pool.js';
       import { makeAppOwnerPool } from './db/app-pool.js';

       const log = pino({ name: 'worker' });
       const redis = new IORedis({

         password: process.env.VALKEY_PASSWORD,

       });
       const litellmPool = makeLitellmPool();
       const appOwnerPool = makeAppOwnerPool();
       const connection = redis as any;
       const queue = createQueue(connection);
       await ensureScheduler(queue);
       const worker = createWorker({ litellmPool, appOwnerPool, connection });
       log.info('worker started');

       async function shutdown() {
         log.info('SIGTERM received, draining…');
         await worker.close();
         await queue.close();
         await litellmPool.end();
         await appOwnerPool.end();
         await redis.quit();
         process.exit(0);
       }
       process.on('SIGTERM', shutdown);
       process.on('SIGINT', shutdown);

<threat_model>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| worker ↔ litellm DB (read) | DIRECT pg connection as openwhispr_owner; reads only |
| worker ↔ openwhispr DB (write) | DIRECT as openwhispr_owner; BYPASSRLS so cross-tenant inserts allowed (worker resolves tenant per row) |
| worker ↔ Valkey | password auth; watermark + BullMQ queue state |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-08-01 | Tampering | Worker uses BYPASSRLS owner role → tenant-mismatched inserts | mitigate | Per-row tenant resolution from users.tenant_id; tests cover tenant correctness; future Phase 6 audit-log will record cross-tenant attempts. |
| T-03-08-02 | Repudiation | Spend rows lost between LiteLLM and ledger if worker crashes mid-batch | mitigate | Watermark only advances after successful insert of the LAST row in the batch; replay safe via ON CONFLICT DO NOTHING. |
| T-03-08-03 | DoS | LiteLLM_SpendLogs grows unbounded → BATCH_SIZE=1000 always lags | accept | v1: 30s cadence + 1000-batch suffices for 1000 concurrent users (rough math). Phase 8 load test validates; Phase 6 may add backlog alert. |
| T-03-08-04 | Information Disclosure | LITELLM_READ_DATABASE_URL contains owner password | mitigate | Same .env handling as DATABASE_URL_OWNER; bootstrap-generated. |
| T-03-08-05 | Spoofing | end_user value is caller-controllable | mitigate | api routes set user = req.user.id (server-side, after dual-auth); end_user in spend logs is therefore authoritative. Phase 4 streaming-token endpoints must preserve this discipline. |
| T-03-08-06 | Cross-DB query failure | Worker accidentally points pool at pgbouncer | mitigate | makeLitellmPool refuses pgbouncer hostname (Pitfall #9 defensive check). |
