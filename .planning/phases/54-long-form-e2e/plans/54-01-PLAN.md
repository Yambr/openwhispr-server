// apps/web/tests/e2e/support/mailpit.ts
export const MAILPIT_BASE: string; // process.env.MAILPIT_API_URL ?? "http://localhost:8025/api/v1"

export interface MailpitFetchOptions {
  /** Lower bound on Created timestamp; messages older than (since - 1s) are skipped. */
  since: Date;
  /** Max wall-clock budget. Default 15_000. */
  timeoutMs?: number;
  /** Poll interval between mailpit list calls. Default 300. */
  pollIntervalMs?: number;
}

/** Returns the absolute verify-email URL extracted from the latest matching
 *  message, or throws after timeoutMs. Matches both /verify-email and
 *  /api/auth/verify-email patterns (Better Auth template flexibility). */
export function fetchVerificationLink(
  email: string,
  opts: MailpitFetchOptions,
): Promise<string>;

/** Same contract, matches /reset-password?token=... URLs. */
export function fetchPasswordResetLink(
  email: string,
  opts: MailpitFetchOptions,
): Promise<string>;

/** DELETE /messages — purges the mailpit inbox. Silently swallows network errors. */
export function clearMessages(): Promise<void>;

<risks>
- **Fake-timer + async fetch races:** vitest's fake timers do NOT auto-advance microtasks the way real timers do. Each `await new Promise(r =&gt; setTimeout(r, pollIntervalMs))` inside the helper needs an explicit `await vi.advanceTimersByTimeAsync(pollIntervalMs)` from the test to unblock. If the test hangs, this is why. Mitigation: use pollIntervalMs=10 in tests and call advanceTimersByTimeAsync in a loop until the assertion resolves.
- **Regex divergence:** the verify-email regex is duplicated in 4 places (100-fullflow inline, signin.steps.ts, this new helper, and Better Auth's email template). Keep the regex BYTE-IDENTICAL to the inline one in 100-fullflow.spec.ts lines 56-58 to avoid silent matching divergence.
- **Mailpit `to:` query case-sensitivity:** the mailpit query API is case-insensitive on the local-part but case-sensitive on the host. The signup helper in fixtures/auth.ts uses `alice+${i}@test.local` (lower); the 100-fullflow spec uses `flow+${ts}@local.test` (lower). The helper does not lowercase the email — if a future caller passes a mixed-case email, mailpit may miss it. Document this in the helper's JSDoc.
</risks>
