   export async function runAxe(page: Page, screenSlug?: string): Promise<void> {
     const results = await new AxeBuilder({ page }).withTags(["wcag2a","wcag2aa","wcag22aa"]).analyze();
     if (screenSlug) {
       const baselinePath = resolve(HERE, "..", "__axe-baselines__", `${screenSlug}.json`);
       if (process.env.AXE_UPDATE_BASELINE === "1") {
         writeFileSync(baselinePath, JSON.stringify(canonicalize(results), null, 2));
         return;
       }
       if (existsSync(baselinePath)) {
         const baseline = JSON.parse(readFileSync(baselinePath, "utf8"));
         const newViolations = diffViolations(results.violations, baseline.violations);
         expect(newViolations).toEqual([]);
         return;
       }
     }
     expect(results.violations).toEqual([]); // fallback: no baseline
   }
