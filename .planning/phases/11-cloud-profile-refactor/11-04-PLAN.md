demo-cloudflared.sh orchestration (kind path):
```bash
#!/usr/bin/env bash
set -euo pipefail
# 1. Boot kind cluster
charts/openwhispr/examples/kind-bootstrap.sh
# 2. Install chart with embedded-litellm + live-demo overlays + operator-supplied OPENAI_API_KEY
helm install openwhispr ./charts/openwhispr \
  -f ./charts/openwhispr/examples/values-embedded-litellm.yaml \
  -f ./charts/openwhispr/examples/values-live-demo.yaml \
  --set secrets.openaiApiKey="${OPENAI_API_KEY:?Set OPENAI_API_KEY first}"
# 3. Wait for api ready
kubectl wait --for=condition=Available deployment/openwhispr-api --timeout=300s
# 4. Port-forward traefik :8443
kubectl port-forward svc/traefik 18443:443 &
PF_PID=$!
trap "kill $PF_PID 2>/dev/null || true; kind delete cluster --name openwhispr-demo 2>/dev/null || true" EXIT
# 5. Run cloudflared pointed at the port-forward
docker run --rm cloudflare/cloudflared:latest tunnel --no-autoupdate --url http://host.docker.internal:18443 | tee /tmp/cf-tunnel.log &
# 6. Poll log for *.trycloudflare.com URL
for i in {1..30}; do
  URL=$(grep -oE 'https://[a-z0-9-]+\.trycloudflare\.com' /tmp/cf-tunnel.log | head -1 || true)
  [[ -n "$URL" ]] && break
  sleep 1
done
[[ -n "$URL" ]] || { echo "Tunnel did not provision a URL within 30s"; exit 1; }
echo "=========================================="
echo "  LIVE DEMO URL: $URL"
echo "=========================================="
echo "Open the URL above in any browser to test."
echo "Ctrl+C to teardown (kind cluster + tunnel)."
wait
```
